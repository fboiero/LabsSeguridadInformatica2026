## Parte 1 — Leone, Lucas (setup, R1, R2)
git checkout -b labextra2-grupo08
./ctf lab extra2
cd labs/labextra2-git-forense/caso/repo-delata
git log --oneline --all
git log --format='%h | %an <%ae> | %ad | %s' --date=iso main
git log -p --reverse main                       # R1: diff de toda la historia
git reflog --date=iso                           # R2: lo que hizo HEAD (incluye el reset)
git fsck --lost-found                           # R2: commit colgado
git fsck --unreachable --no-reflogs             # R2: commit + tree + blob sin referencia
git cat-file -t bc9d0db
git cat-file -p bc9d0db
git show bc9d0db                                # R2: contenido del commit "borrado"
cd ../../../..
./ctf submit extra2 R1 'FLAG{git_log_no_olvida}'
./ctf submit extra2 R2 'FLAG{git_reflog_rescata}'
./ctf status extra2
# salidas crudas -> evidencia/salidas/r1.txt y r2.txt (redirigidas con >)
